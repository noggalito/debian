# Plantilla de /etc/audit/rules.d/99-hardening.rules v1.1
# Reglas mínimas de auditoría
#
# Cómo se construye el archivo que realmente se carga:
#   augenrules concatena todos los *.rules de /etc/audit/rules.d/ en orden
#   natural y escribe el resultado en /etc/audit/audit.rules, descartando
#   líneas vacías y comentarios. Estos comentarios, por tanto, no llegan al
#   archivo fusionado: viven solo aquí.
#
#   Las cuatro directivas globales no se quedan donde uno las escribe.
#   augenrules reubica el último -D como primera línea, el último -b como
#   segunda, el último -f como tercera y el último -e como última. "Último"
#   se refiere al orden de concatenación, no a la posición dentro del
#   archivo. En orden natural este archivo va ANTES que el audit.rules que
#   trae el paquete auditd, así que para esas cuatro directivas gana el
#   valor de Debian, no el de aquí. Comprobar el resultado efectivo con:
#       sudo augenrules --check
#       head -4 /etc/audit/audit.rules ; tail -1 /etc/audit/audit.rules
#       sudo auditctl -s

# Descartar cualquier regla previa para partir de un estado conocido.
# augenrules lo promueve a la primera línea del archivo fusionado, así que
# no elimina lo que declaren los demás componentes: solo limpia el kernel
# antes de cargar el conjunto completo.
-D

# Buffer de eventos (mayor = menos pérdidas bajo carga)
-b 8192

# Modo de fallo ante pérdida de eventos: 0 silencioso, 1 printk, 2 panic.
# Se declara aquí para que el archivo sea autocontenido, aunque el paquete
# auditd de Debian ya fija el mismo valor.
-f 1

# === Identidad y privilegios ===
-w /etc/passwd       -p wa -k identidad
-w /etc/shadow       -p wa -k identidad
-w /etc/group        -p wa -k identidad
-w /etc/gshadow      -p wa -k identidad
-w /etc/sudoers      -p wa -k privilegios
-w /etc/sudoers.d/   -p wa -k privilegios

# === SSH ===
-w /etc/ssh/sshd_config       -p wa -k ssh
-w /etc/ssh/sshd_config.d/    -p wa -k ssh

# === Configuración crítica ===
-w /etc/hosts        -p wa -k red
-w /etc/resolv.conf  -p wa -k red
-w /etc/sysctl.conf  -p wa -k sysctl
-w /etc/sysctl.d/    -p wa -k sysctl

# === Escalada de privilegios ===
-a always,exit -F arch=b64 -S execve -F euid=0 -F auid>=1000 -F auid!=4294967295 -k root_exec
-a always,exit -F arch=b32 -S execve -F euid=0 -F auid>=1000 -F auid!=4294967295 -k root_exec

# Modo inmutable: una vez cargado este conjunto, el kernel rechaza cualquier
# cambio de reglas hasta el próximo reinicio. Es la razón por la que, después
# de aplicar esta plantilla, un `augenrules --load` posterior falla y
# `auditctl -D` no vacía nada: no es un problema de permisos, es la
# protección funcionando. Para modificar reglas hay que editar el archivo y
# reiniciar el servidor.
#
# Alternativa si se prefiere poder recargar sin reiniciar: -e 1. El costo es
# que un atacante con root puede desactivar la auditoría en caliente.
#
# La posición no importa: augenrules emite el último -e como última línea del
# archivo fusionado.
-e 2
